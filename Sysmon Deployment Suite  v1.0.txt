function Get-ConfigFile {
    Write-Host "`n  [1/3] Downloading Sysmon configuration..." -ForegroundColor Cyan
    foreach ($src in $ConfigSources) {
        try {
            Write-Host ("       → Trying: {0}" -f $src.Name) -ForegroundColor DarkGray
            Invoke-WebRequest -Uri $src.Url -OutFile $ConfigPath -UseBasicParsing -ErrorAction Stop
            if ((Test-Path $ConfigPath) -and (Get-Item $ConfigPath).Length -gt 0) {
                Write-Host ("       ✔ Config OK ({0:N0} bytes) from {1}" -f (Get-Item $ConfigPath).Length, $src.Name) -ForegroundColor Green
                return $true
            }
        } catch {
            Write-Host "       ✘ Not available, trying next..." -ForegroundColor DarkYellow
        }
    }
    Write-Host "  ✘ ERROR: All config sources failed." -ForegroundColor Red
    return $false
}

function Invoke-SysmonRun {
    Show-Banner
    Write-Host "  ▶  RUN MODE — Installing / Updating Sysmon" -ForegroundColor Green
    Write-Host "  ──────────────────────────────────────────────" -ForegroundColor DarkGray

    # 1. Get config
    if (-not (Get-ConfigFile)) { Pause-Return; return }

    # 2. Uninstall old
    Write-Host "`n  [2/3] Checking for existing Sysmon..." -ForegroundColor Cyan
    if (Test-Path "C:\Windows\Sysmon64.exe") {
        Write-Host "       → Uninstalling existing Sysmon..." -ForegroundColor Yellow
        & "C:\Windows\Sysmon64.exe" -u | Out-Null
        Start-Sleep -Seconds 2
    } else {
        Write-Host "       → No existing Sysmon found." -ForegroundColor DarkGray
    }

    # 3. Download & extract Sysmon
    Write-Host "`n  [3/3] Deploying Sysmon v15.22..." -ForegroundColor Cyan
    Write-Host "       → Downloading binaries..." -ForegroundColor DarkGray
    Invoke-WebRequest -Uri $SysmonUrl -OutFile $ZipPath -UseBasicParsing
    Write-Host "       → Extracting..." -ForegroundColor DarkGray
    Expand-Archive -Path $ZipPath -DestinationPath $ExtractPath -Force
    Write-Host "       → Installing with configuration..." -ForegroundColor DarkGray
    & "$ExtractPath\Sysmon64.exe" -accepteula -i $ConfigPath

    # 4. Verify
    Start-Sleep -Seconds 3
    $svc = Get-Service Sysmon64 -ErrorAction SilentlyContinue
    if ($svc -and $svc.Status -eq "Running") {
        Remove-Item $ZipPath, $ExtractPath -Recurse -Force -ErrorAction SilentlyContinue
        Write-Host ""
        Write-Host "  ╔════════════════════════════════════════════════════╗" -ForegroundColor Green
        Write-Host "  ║  ✔  SUCCESS — Sysmon installed & running           ║" -ForegroundColor Green
        Write-Host "  ║  Config  : Olaf Hartong / SwiftOnSecurity          ║" -ForegroundColor Green
        Write-Host "  ║  Event 7 : Image Load        → ENABLED             ║" -ForegroundColor Green
        Write-Host "  ║  Event 10: Process Access    → ENABLED             ║" -ForegroundColor Green
        Write-Host "  ╚════════════════════════════════════════════════════╝" -ForegroundColor Green
    } else {
        Write-Host ""
        Write-Host "  ✘ FAILED: Sysmon service is NOT running." -ForegroundColor Red
    }
    Pause-Return
}

function Invoke-SysmonStatus {
    Show-Banner
    Write-Host "  ✔  STATUS — Sysmon Service State" -ForegroundColor Cyan
    Write-Host "  ──────────────────────────────────────────────" -ForegroundColor DarkGray
    Write-Host ""

    $svc = Get-Service Sysmon64 -ErrorAction SilentlyContinue
    if ($svc) {
        Write-Host ("  Service Name   : {0}" -f $svc.Name)
        Write-Host ("  Display Name   : {0}" -f $svc.DisplayName)
        Write-Host ("  Status         : {0}" -f $svc.Status) -ForegroundColor Green
        Write-Host ("  Start Type     : {0}" -f $svc.StartType)

        $drv = Get-Service SysmonDrv -ErrorAction SilentlyContinue
        if ($drv) {
            Write-Host ("  Driver Status  : {0}" -f $drv.Status) -ForegroundColor Green
        }
        if (Test-Path "C:\Windows\Sysmon64.exe") {
            $ver = (Get-Item "C:\Windows\Sysmon64.exe").VersionInfo.FileVersion
            Write-Host ("  Binary Version : {0}" -f $ver)
        }
    } else {
        Write-Host "  ✘ Sysmon is NOT installed." -ForegroundColor Red
    }
    Pause-Return
}

function Invoke-SysmonReconfigure {
    Show-Banner
    Write-Host "  ⚙  RECONFIGURE — Reload config on running Sysmon" -ForegroundColor Cyan
    Write-Host "  ──────────────────────────────────────────────" -ForegroundColor DarkGray

    $svc = Get-Service Sysmon64 -ErrorAction SilentlyContinue
    if (-not $svc) {
        Write-Host "  ✘ Sysmon not installed. Use [1] RUN first." -ForegroundColor Red
        Pause-Return; return
    }
    if (-not (Get-ConfigFile)) { Pause-Return; return }

    Write-Host "`n  → Applying config..." -ForegroundColor Cyan
    & "C:\Windows\Sysmon64.exe" -c $ConfigPath
    Write-Host "  ✔ Reconfiguration command completed." -ForegroundColor Green
    Pause-Return
}

function Invoke-SysmonUninstall {
    Show-Banner
    Write-Host "  🗑  UNINSTALL — Remove Sysmon from this system" -ForegroundColor Yellow
    Write-Host "  ──────────────────────────────────────────────" -ForegroundColor DarkGray

    if (Test-Path "C:\Windows\Sysmon64.exe") {
        Write-Host "  → Uninstalling Sysmon..." -ForegroundColor Yellow
        & "C:\Windows\Sysmon64.exe" -u
        Write-Host "  ✔ Sysmon uninstalled." -ForegroundColor Green
    } else {
        Write-Host "  ✘ Sysmon is not installed." -ForegroundColor DarkGray
    }
    Pause-Return
}

function Invoke-SysmonEvents {
    Show-Banner
    Write-Host "  📜  EVENTS — Latest Sysmon Events (7 & 10 focus)" -ForegroundColor Cyan
    Write-Host "  ──────────────────────────────────────────────" -ForegroundColor DarkGray

    try {
        Write-Host "`n  Last 10 events from Sysmon log:`n" -ForegroundColor Cyan
        Get-WinEvent -LogName "Microsoft-Windows-Sysmon/Operational" -MaxEvents 10 -ErrorAction Stop |
            Select-Object TimeCreated, Id, LevelDisplayName |
            Format-Table -AutoSize

        Write-Host "  Event ID 7 & 10 counts (last 500 events):" -ForegroundColor Cyan
        Get-WinEvent -LogName "Microsoft-Windows-Sysmon/Operational" -MaxEvents 500 -ErrorAction SilentlyContinue |
            Where-Object { $_.Id -in 7,10 } |
            Group-Object Id | Select-Object @{n='EventID';e={$_.Name}}, Count | Format-Table -AutoSize
    } catch {
        Write-Host "  ✘ Could not read Sysmon event log: $($_.Exception.Message)" -ForegroundColor Red
    }
    Pause-Return
}
